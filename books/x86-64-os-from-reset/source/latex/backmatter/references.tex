\chapter*{规范与延伸阅读}
\addcontentsline{toc}{chapter}{规范与延伸阅读}
\markboth{规范与延伸阅读}{规范与延伸阅读}

\begin{enumerate}
  \item BIPM, \emph{The International System of Units (SI Brochure)}，
  秒、赫兹、伏特、安培、欧姆、瓦特与十进制前缀的统一定义。
  \item Paul Horowitz and Winfield Hill, \emph{The Art of Electronics}，
  电阻、电容、二极管、晶体管、供电、去耦、噪声与实验测量的工程直觉。
  \item Adel S. Sedra and Kenneth C. Smith, \emph{Microelectronic Circuits}，
  半导体器件、MOSFET、CMOS 反相器、模拟范围与数字开关基础。
  \item David Money Harris and Sarah L. Harris,
  \emph{Digital Design and Computer Architecture}，布尔逻辑、组合/时序电路、
  加法器、状态机、处理器数据通路、存储器和接口。
  \item David A. Patterson and John L. Hennessy,
  \emph{Computer Organization and Design}，ISA 与微架构、取指执行、流水线、
  存储层次和 I/O 的体系化连接。
  \item JEDEC, \emph{Terms, Definitions, and Letter Symbols for
  Microelectronic Devices} 及相关存储器标准，SRAM、DRAM、非易失存储与
  器件时序术语。
  \item Intel, \emph{Intel 64 and IA-32 Architectures Software Developer's
  Manual}, Version 092, Volume 3A。正文重点对照 5.5.4 的四/五级地址翻译、
  6.8.8 的 64 位快速系统调用、7.14.2--7.14.5 的中断现场与 IST、
  12.1.4 的复位首指令和 12.8.5 的 IA-32e 初始化顺序；书中相关图均按
  字段关系重新绘制。
  \item Intel, \emph{Intel 64 and IA-32 Architectures Software Developer's
  Manual}, Version 092, Volume 3B，23.1--23.1.4 的实地址模式环境、
  段地址翻译、可用寄存器、指令类别和 IVT 中断现场。第 5 章的地址图、指令
  分类和中断表均按这些字段重新绘制，并与项目 Stage 1 代码逐项对应。
  \item Intel, \emph{Intel 64 and IA-32 Architectures Software Developer's
  Manual}, Version 092, Volumes 1 and 2，处理器模式、寄存器、指令编码和
  逐条指令语义。
  \item Intel, \emph{Intel Processor and Intel Core i3 and Intel Core 3
  N-Series Datasheet, Volume 1 of 2}，Document 759603 Rev.002，以及随附的
  \code{759603\_001\_Ballout.xlsx} Rev.001。第 6 章的 BGA1264 尺寸、
  1264 个球位坐标、信号名和接口分类据此重绘；生成时使用的 Ballout 文件
  SHA-256 为
  \code{25c6df23fccf4a9c3859787764e35ff8fe1d0ae839509d5ee9b3130a5bee9bd8}。
  \item AMD, \emph{AMD64 Architecture Programmer's Manual}, Volume 2，
  Revision 3.44，长模式及其 64 位/兼容子模式、系统寄存器、异常、中断和
  页表转换；“long mode”采用 AMD 规范中的模式名称。
  \item System V ABI, \emph{AMD64 Architecture Processor Supplement}，
  C++ 与汇编边界、寄存器归属、栈对齐、ELF64 重定位和进程初始状态。
  \item QEMU Project, \emph{System Emulation User's Guide}，机器、固件、
  TCG、字符设备和块设备参数。
  \item QEMU Project, \emph{fw\_cfg specification}，selector/data
  传输、文件目录、大端字段与来宾配置发现。
  \item IEEE/The Open Group, \emph{ELF gABI}，ELF 文件头、程序头和装载段。
  \item INCITS, \emph{ATA/ATAPI Command Set}，ATA PIO 端口、状态机、
  LBA 寻址和错误返回。
  \item National Semiconductor, \emph{PC16550D Universal
  Asynchronous Receiver/Transmitter With FIFOs}，串口寄存器与状态位。
  \item Intel, \emph{8259A Programmable Interrupt Controller} 与
  \emph{8254 Programmable Interval Timer}，传统 PC 中断路由、EOI、
  计数器模式和时钟分频。
  \item Intel, \emph{8042 Keyboard Controller}，控制器状态、命令字节和
  PS/2 键盘数据通路。
  \item ISO/IEC 14882:2020, \emph{Programming Languages --- C++}，
  C++20 对象模型、类型系统、初始化与 freestanding 实现边界。
  \item Linux Kernel Documentation, \emph{Memory Map on x86-64}，
  四级/五级虚拟地址布局、direct-map 区间与实现边界。项目只借鉴分层方向，
  不宣称 Linux ABI 或内部实现兼容。
  \item Jeff Bonwick, \emph{The Slab Allocator: An Object-Caching Kernel
  Memory Allocator}, USENIX Summer 1994，固定大小内核对象缓存、对象复用、
  着色和调试能力的历史来源。
  \item Jeff Bonwick and Jonathan Adams, \emph{Magazines and Vmem:
  Extending the Slab Allocator to Many CPUs and Arbitrary Resources},
  USENIX 2001，多处理器本地缓存、分层资源分配与未来扩展方向。
  \item UEFI Forum, \emph{Unified Extensible Firmware Interface
  Specification}，PE/COFF 启动入口、Boot Services、Memory Map、
  ExitBootServices 与操作系统交接。
  \item UEFI Forum, \emph{Advanced Configuration and Power Interface
  Specification}，RSDP、MADT、MCFG、平台设备、电源状态与操作系统发现。
  \item PCI-SIG, \emph{PCI Express Base Specification} 与
  \emph{Card Electromechanical Specification}，lane、链路训练、参考时钟、
  配置空间、BAR、插槽电源和控制信号。
  \item USB-IF, \emph{Universal Serial Bus Specification}、
  \emph{USB Type-C Cable and Connector Specification} 与
  \emph{USB Power Delivery Specification}，D+/D-、SuperSpeed、
  枚举、Type-C CC 和 PD 协商。
  \item NXP Semiconductors, \emph{UM10204 I2C-bus specification and
  user manual}，开漏、上拉、起始/停止、寻址、应答和时序。
  \item LattePandaTeam, \emph{LattePanda Mu Hardware Design Files}，
  DFR1142 Lite Carrier V2 的 KiCad 9 十页原理图与四层 PCB。本书固定到
  上游提交 \code{f954bf0275fa0aec4c1e9eb168f09644563b28a4}，MIT 许可；
  本地归档位于
  \filepath{hardware/lattepanda_mu_learning_carrier/upstream_dfr1142_v2/}。
  \item LattePandaTeam, \emph{LattePanda Mu Edge Connector Pinout}，
  260-pin 电源、HSIO、USB、DDI、I2C、UART、SPI、风扇和控制信号；
  本地表格与引脚图位于
  \filepath{hardware/lattepanda_mu_learning_carrier/module_pinout/}。
  \item KiCad Project, \emph{KiCad 9 Documentation}，层次原理图、网络、
  footprint 映射、PCB 叠层、网络类别、ERC 与 DRC。自动检查是设计门禁，
  不替代器件、电源、信号完整性、热与制造审查。
  \item 本项目的
  \filepath{hardware/lattepanda_mu_learning_carrier/provenance.md} 与
  \filepath{hardware/lattepanda_mu_learning_carrier/manufacturing_checklist.md}，
  记录实体案例来源、哈希、许可、当前工具限制和投板前必须重新确认的事项。
  \item 本项目的
  \filepath{docs/learning/physical-carrier-circuit-guide.md}、
  \filepath{tools/generate_physical_carrier_diagrams.py} 与
  \filepath{tools/check_learning_diagrams.py}，用于重复生成三张学习图并
  验证可见引脚、网络、导线连通分量、NC 与安全区。
  \item 随书配套源码中的 \filepath{docs/adr/}、\filepath{docs/architecture.md}
  和 \filepath{docs/releases/}，用于追踪实现决策与阶段证据。
\end{enumerate}
