\chapter*{规范与延伸阅读}
\addcontentsline{toc}{chapter}{规范与延伸阅读}

\begin{enumerate}
  \item Intel, \emph{Intel 64 and IA-32 Architectures Software Developer's
  Manual}, Volume 3A，处理器初始化、控制寄存器、保护模式和分页。
  \item Intel, \emph{Intel 64 and IA-32 Architectures Software Developer's
  Manual}, Volume 2，指令编码和语义。
  \item AMD, \emph{AMD64 Architecture Programmer's Manual}, Volume 2，
  长模式、系统寄存器、异常、中断和页表转换。
  \item System V ABI, \emph{AMD64 Architecture Processor Supplement}，
  C++ 与汇编边界、寄存器归属、栈对齐、ELF64 重定位和进程初始状态。
  \item QEMU Project, \emph{System Emulation User's Guide}，机器、固件、
  TCG、字符设备和块设备参数。
  \item QEMU Project, \emph{fw\_cfg specification}，selector/data
  传输、文件目录、大端字段与来宾配置发现。
  \item IEEE/The Open Group, \emph{ELF gABI}，ELF 文件头、程序头和装载段。
  \item INCITS, \emph{ATA/ATAPI Command Set}，ATA PIO 端口、状态机、
  LBA 寻址和错误返回。
  \item National Semiconductor, \emph{PC16550D Universal
  Asynchronous Receiver/Transmitter With FIFOs}，串口寄存器与状态位。
  \item Intel, \emph{8259A Programmable Interrupt Controller} 与
  \emph{8254 Programmable Interval Timer}，传统 PC 中断路由、EOI、
  计数器模式和时钟分频。
  \item Intel, \emph{8042 Keyboard Controller}，控制器状态、命令字节和
  PS/2 键盘数据通路。
  \item ISO/IEC 14882:2020, \emph{Programming Languages --- C++}，
  C++20 对象模型、类型系统、初始化与 freestanding 实现边界。
  \item Linux Kernel Documentation, \emph{Memory Map on x86-64}，
  四级/五级虚拟地址布局、direct-map 区间与实现边界。项目只借鉴分层方向，
  不宣称 Linux ABI 或内部实现兼容。
  \item Jeff Bonwick, \emph{The Slab Allocator: An Object-Caching Kernel
  Memory Allocator}, USENIX Summer 1994，固定大小内核对象缓存、对象复用、
  着色和调试能力的历史来源。
  \item Jeff Bonwick and Jonathan Adams, \emph{Magazines and Vmem:
  Extending the Slab Allocator to Many CPUs and Arbitrary Resources},
  USENIX 2001，多处理器本地缓存、分层资源分配与未来扩展方向。
  \item 配套仓库的 \filepath{docs/adr/}、\filepath{docs/architecture.md}
  和 \filepath{docs/releases/}，用于追踪实现决策与阶段证据。
\end{enumerate}
